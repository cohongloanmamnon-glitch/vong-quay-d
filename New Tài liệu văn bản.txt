// TỔNG HỢP ÂM THANH BẰNG WEB AUDIO API
const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

function playSound(type) {
  if (!isSoundOn) return;
  try {
    if (audioCtx.state === 'suspended') {
      audioCtx.resume();
    }
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);

    if (type === 'spin') {
      osc.type = 'triangle';
      osc.frequency.setValueAtTime(300, audioCtx.currentTime);
      osc.frequency.exponentialRampToValueAtTime(150, audioCtx.currentTime + 0.08);
      gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
      gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.08);
      osc.start();
      osc.stop(audioCtx.currentTime + 0.08);
    } else if (type === 'correct') {
      osc.type = 'sine';
      osc.frequency.setValueAtTime(523.25, audioCtx.currentTime);
      osc.frequency.setValueAtTime(659.25, audioCtx.currentTime + 0.1);
      osc.frequency.setValueAtTime(783.99, audioCtx.currentTime + 0.2);
      gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
      gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);
      osc.start();
      osc.stop(audioCtx.currentTime + 0.4);
    } else if (type === 'wrong') {
      osc.type = 'sawtooth';
      osc.frequency.setValueAtTime(200, audioCtx.currentTime);
      osc.frequency.setValueAtTime(140, audioCtx.currentTime + 0.15);
      gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
      gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
      osc.start();
      osc.stop(audioCtx.currentTime + 0.3);
    } else if (type === 'win') {
      osc.type = 'sine';
      osc.frequency.setValueAtTime(440, audioCtx.currentTime);
      osc.frequency.setValueAtTime(554.37, audioCtx.currentTime + 0.15);
      osc.frequency.setValueAtTime(659.25, audioCtx.currentTime + 0.3);
      osc.frequency.setValueAtTime(880, audioCtx.currentTime + 0.45);
      gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
      gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.7);
      osc.start();
      osc.stop(audioCtx.currentTime + 0.7);
    }
  } catch (e) {
    console.log("Audio play error", e);
  }
}

// PHÁT ÂM TIẾNG VIỆT (WEB SPEECH SYNTHESIS)
function speakText(text) {
  if (!isSoundOn || !('speechSynthesis' in window)) return;
  window.speechSynthesis.cancel();
  const utter = new SpeechSynthesisUtterance(text);
  utter.lang = 'vi-VN';
  utter.rate = 0.85;
  window.speechSynthesis.speak(utter);
}

// VẼ VÒNG QUAY BẰNG CANVAS

function drawWheel() {
  const numSlices = VOCAB_ITEMS.length;
  const arc = (2 * Math.PI) / numSlices;
  const centerX = canvas.width / 2;
  const centerY = canvas.height / 2;
  const radius = canvas.width / 2 - 10;

  ctx.clearRect(0, 0, canvas.width, canvas.height);

  for (let i = 0; i < numSlices; i++) {
    const angle = i * arc;
    
    // Vẽ múi tròn
    ctx.beginPath();
    ctx.fillStyle = SLICE_COLORS[i % SLICE_COLORS.length];
    ctx.moveTo(centerX, centerY);
    ctx.arc(centerX, centerY, radius, angle, angle + arc);
    ctx.lineTo(centerX, centerY);
    ctx.fill();

    // Viền múi
    ctx.strokeStyle = "rgba(255,255,255,0.6)";
    ctx.lineWidth = 4;
    ctx.stroke();

    // Lên chữ trên múi
    ctx.save();
    ctx.translate(centerX, centerY);
    ctx.rotate(angle + arc / 2);
    ctx.textAlign = "right";
    ctx.fillStyle = "#FFFFFF";
    ctx.font = "bold 24px 'Be Vietnam Pro', sans-serif";
    ctx.shadowColor = "rgba(0,0,0,0.4)";
    ctx.shadowBlur = 4;
    ctx.fillText(VOCAB_ITEMS[i].word, radius - 30, 8);
    ctx.restore();
  }

  // Vẽ viền ngoài vòng quay
  ctx.beginPath();
  ctx.arc(centerX, centerY, radius, 0, 2 * Math.PI);
  ctx.strokeStyle = "#FFD45A";
  ctx.lineWidth = 12;
  ctx.stroke();
}

// =====================================================
// XỬ LÝ QUAY VÒNG QUAY
// =====================================================
function spinWheel() {
  if (isSpinning) return;

  // Lựa chọn ô chưa hoàn thành nếu còn
  let uncompletedIndices = [];
  itemStatus.forEach((done, index) => {
    if (!done) uncompletedIndices.push(index);
  });

  let selectedIndex;
  if (uncompletedIndices.length > 0) {
    selectedIndex = uncompletedIndices[Math.floor(Math.random() * uncompletedIndices.length)];
  } else {
    selectedIndex = Math.floor(Math.random() * VOCAB_ITEMS.length);
  }

  isSpinning = true;
  spinBtn.disabled = true;
  instruction.innerText = "🎡 vòng quay đang quay...";
  feedback.innerText = "";
  feedback.className = "feedback";
  nextBtn.classList.add("hidden");

  const numSlices = VOCAB_ITEMS.length;
  const sliceArc = 360 / numSlices;

  // Tính toán góc quay để mũi tên dừng chính xác ở đỉnh (270 deg)
  const targetSliceCenter = selectedIndex * sliceArc + sliceArc / 2;
  const targetAngleInWheel = 270 - targetSliceCenter;
  
  const extraRounds = 5 + Math.floor(Math.random() * 3);
  const currentMod = currentRotation % 360;
  let addedRotation = (targetAngleInWheel - currentMod) % 360;
  if (addedRotation < 0) addedRotation += 360;

  currentRotation += extraRounds * 360 + addedRotation;
  canvas.style.transform = `rotate(${currentRotation}deg)`;

  // Âm thanh khi quay
  let spinAudioInterval = setInterval(() => {
    playSound('spin');
  }, 180);

  setTimeout(() => {
    clearInterval(spinAudioInterval);
    isSpinning = false;
    spinBtn.disabled = false;
    showChallenge(selectedIndex);
  }, 4000);
}

// HIỂN THỊ THỬ THÁCH BÀI HỌC
// =====================================================
function showChallenge(index) {
  currentItem = VOCAB_ITEMS[index];
  
  // Hiển thị hình ảnh từ tập tin ảnh
  pictureContainer.innerHTML = `<img src="${currentItem.image}" alt="${currentItem.word}">`;
  pictureContainer.classList.add("pop");
  setTimeout(() => pictureContainer.classList.remove("pop"), 300);

  wordTitle.innerText = currentItem.word;
  promptText.innerHTML = `bé hãy tìm chữ <b>"${currentItem.target}"</b> trong từ bên trên!`;
  instruction.innerText = `👇 chạm vào chữ ${currentItem.target.toUpperCase()} để chọn!`;

  speakText(`Bé hãy tìm chữ ${currentItem.target} trong từ ${currentItem.word}`);

  // Tạo các nút chữ cái
  lettersContainer.innerHTML = "";
  const chars = Array.from(currentItem.word);

  chars.forEach((ch, idx) => {
    if (ch === ' ') {
      const spaceDiv = document.createElement("div");
      spaceDiv.className = "letter space";
      lettersContainer.appendChild(spaceDiv);
    } else {
      const letterBtn = document.createElement("button");
      letterBtn.className = "letter";
      letterBtn.innerText = ch;
      letterBtn.dataset.char = ch.toLowerCase();
      letterBtn.dataset.index = idx;

      letterBtn.addEventListener("click", () => handleLetterClick(letterBtn, ch.toLowerCase()));
      lettersContainer.appendChild(letterBtn);
    }
  });
}

// =====================================================
// XỬ LÝ KHI BÉ BẤM CHỌN CHỮ CÁI
// =====================================================
function handleLetterClick(buttonEl, clickedChar) {
  if (buttonEl.classList.contains("correct") || buttonEl.classList.contains("wrong")) return;

  const targetChar = currentItem.target.toLowerCase();

  if (clickedChar === targetChar) {
    buttonEl.classList.add("correct");
    playSound('correct');
    
    score += 10;
    scoreEl.innerText = score;

    feedback.innerText = `👏 hoan hô! bé đã chọn đúng chữ "${targetChar}"!`;
    feedback.className = "feedback pop";
    feedback.style.background = "#bdf4cf";
    feedback.style.color = "#16803b";

    speakText(`Đúng rồi! Chữ ${targetChar}`);

    const currentIndex = VOCAB_ITEMS.indexOf(currentItem);
    if (currentIndex !== -1 && !itemStatus[currentIndex]) {
      itemStatus[currentIndex] = true;
      completedCount++;
    }

    const allButtons = lettersContainer.querySelectorAll(".letter:not(.space)");
    allButtons.forEach(btn => {
      if (btn.dataset.char === targetChar) {
        btn.classList.add("correct");
      }
    });

    nextBtn.classList.remove("hidden");
    instruction.innerText = "🌟 giỏi lắm! nhấn 'quay tiếp' nhé!";

    if (completedCount >= VOCAB_ITEMS.length) {
      setTimeout(() => {
        playSound('win');
        overlay.classList.add("show");
        speakText("Xuất sắc quá! Bé đã tìm đúng tất cả các từ!");
      }, 1200);
    }

  } else {
    buttonEl.classList.add("wrong");
    playSound('wrong');

    feedback.innerText = `😅 chưa đúng rồi, bé hãy tìm chữ "${targetChar}" nhé!`;
    feedback.className = "feedback pop";
    feedback.style.background = "#ffd4d8";
    feedback.style.color = "#e83b52";

    speakText(`Thử lại nào, bé tìm chữ ${targetChar} nhé`);

    setTimeout(() => {
      buttonEl.classList.remove("wrong");
    }, 800);
  }
}
